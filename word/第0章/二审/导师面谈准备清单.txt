代码内容：
─────────────────────────────────────────────────────────────────
import torch
import torch.nn.functional as F
from torchvision import transforms
from PIL import Image
from model import ImageRetrievalModel


def load_model(checkpoint_path="./checkpoints/best_model.pth", device="cpu"):
    model = ImageRetrievalModel(
        backbone="resnet50",
        feature_dim=512,
        hash_bits=128,
        use_multimodal=True,  # 论文主实验是多模态
        pretrained=False
    )
    state = torch.load(checkpoint_path, map_location=device)
    model.load_state_dict(state)
    model.eval().to(device)
    return model


def preprocess(image_path):
    transform = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
    ])
    img = Image.open(image_path).convert("RGB")
    return transform(img).unsqueeze(0)


def infer(image_path, lat=None, lon=None, model=None, device="cpu"):
    img_tensor = preprocess(image_path).to(device)
    loc = None
    if lat is not None and lon is not None:
        loc = torch.tensor([[lat, lon]], dtype=torch.float32).to(device)

    with torch.no_grad():
        hash_continuous = model(img_tensor, loc)  # Tanh输出 [-1,1]
        hash_binary = (hash_continuous > 0).float()  # {0,1}

    return hash_continuous.cpu().numpy(), hash_binary.cpu().numpy()


if __name__ == "__main__":
    model = load_model(device="cpu")
    # 用任意一张本地图片测试
    h_cont, h_bin = infer("test.jpg", lat=39.9289, lon=116.3974, model=model)
    print("连续哈希码:", h_cont.shape, h_cont)
    print("二进制哈希码:", h_bin.shape, h_bin)
    print("码值:", h_bin[0][:16], "...")

Step 1：预计算结果
  用 inference.py 对 BFATH 演示集的 8 张图逐一推理，生成预计算数据库：

  # algorithm/precompute.py
  import json, os
  from inference import load_model, infer

  model = load_model(device="cpu")
  db = []
  for item in os.listdir("data_samples/bfath_demo"):
      path = f"data_samples/bfath_demo/{item}/image.jpg"
      meta = json.load(open(f"data_samples/bfath_demo/{item}/meta.json"))
      _, h_bin = infer(path, lat=meta["lat"], lon=meta["lon"], model=model)
      db.append({
          "id": item.split("_")[0],
          "name": meta["name"],
          "species": meta["species"],
          "age": meta["age"],
          "lat": meta["lat"], "lon": meta["lon"],
          "hash_code": h_bin[0].tolist()  # 128位列表
      })
  json.dump(db, open("precomputed_db.json", "w", encoding="utf-8"),
            ensure_ascii=False, indent=2)

【六、P4：导师最可能问的 5 个问题及回复模板】
═══════════════════════════════════════════════════════════════════

Q1：模型名 Adaptive，但实验全用拼接，怎么解释？
─────────────────────────────────────────────────────────────────
答："Adaptive 体现在两个层面：第一，模型架构支持自适应权重融合——
    代码里实现了 MultiHeadAttentionFusion（model.py:408），
    论文 3.2.3 节公式 3-8、3-9 也给出了自适应权重的数学形式；
    第二，受限于 BFATH 数据集规模只有 300 张，引入 Adaptive 的注意力权重
    矩阵会增加过拟合风险，因此选择更稳健的特征拼接作为主实验方案。
    这是工程取舍，不是技术能力缺失。"

Q2："以图搜位置"到底是 Geo-localization 还是数据库检索？
─────────────────────────────────────────────────────────────────
答："DHLAM 本质上是数据库检索，不是 Geo-localization——给定查询图像，
    模型在数据库中检索视觉相似的古树记录，并返回这些记录的已知地理位置。
    系统不具备预测任意新位置的能力，只是返回已有记录的坐标。"

Q3：BFATH 只有 300 张，消融实验可信吗？
─────────────────────────────────────────────────────────────────
答："BFATH 是领域验证数据集，主要观察各模块的相对贡献趋势。
    核心性能对比是在 12,450 张的 iNaturalist 2018 上完成（表 3.3），
    大样本结论统计稳健。BFATH 上的精确数值受样本量限制存在波动，
    但贡献排序（地理位置 > 难例挖掘 > 量化损失）是稳定的。"

Q4：为什么只选 3 个单模态基线，没有多模态基线？
─────────────────────────────────────────────────────────────────
答："现有多模态哈希方法（如 DCMH、SSAH）的文本编码器针对离散文本 token
    设计，不适用于连续的地理位置坐标。重新设计位置编码器并训练多模态基线
    的工程量较大，留作未来工作。本文通过构建 DHLAM-Image 单模态变体
    作为公平对比基准。"

Q5：模型怎么保证和论文描述一致？现场能跑吗？
─────────────────────────────────────────────────────────────────
答："模型结构和超参数全部在 config.py 中集中配置，与论文表 3.1 一一对应。
    我现在可以现场演示：输入一张古树图像和坐标，模型输出 128 位二进制哈希码。"
    （然后现场打开 inference.py 执行）
